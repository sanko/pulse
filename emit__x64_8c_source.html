248 uint8_t mod = (offset == 0 && (base & 0x07) != 5) ? 0x00 : ((offset >= -128 && offset <= 127) ? 0x40 : 0x80);
249 EMIT_CHECK(emit_x64_rex(ctx, true, EMIT_REG_NEEDS_REX(dest), false, EMIT_REG_NEEDS_REX(base)));
250 EMIT_CHECK(emit_emit_u8(ctx, 0x8B));

252 if ((base & 0x07) == 4)
253 EMIT_CHECK(emit_emit_u8(ctx, 0x24));
254 if (mod == 0x40)
255 EMIT_CHECK(emit_emit_u8(ctx, (uint8_t)offset));
