47static uint8_t x64_to_arm64_cc(emit_cc_t cc) {
48 switch (cc) {
49 case EMIT_CC_E:
50 return 0x0;
51 case EMIT_CC_NE:
52 return 0x1;
53 case EMIT_CC_B:
54 return 0x2;
55 case EMIT_CC_AE:
56 return 0x3;
57 case EMIT_CC_S:
58 return 0x4;
59 case EMIT_CC_NS:
60 return 0x5;
61 case EMIT_CC_P:
62 return 0x6;
63 case EMIT_CC_NP:
64 return 0x7;
65 case EMIT_CC_A:
66 return 0x8;
67 case EMIT_CC_BE:
68 return 0x9;
69 case EMIT_CC_GE:
70 return 0xA;
71 case EMIT_CC_L:
72 return 0xB;
73 case EMIT_CC_G:
74 return 0xC;
75 case EMIT_CC_LE:
76 return 0xD;
77 case EMIT_CC_AL:
78 case EMIT_CC_NOCARE:
79 default:
80 return 0xE;
81 }
82}
