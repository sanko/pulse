79symbol_t * sema_lookup(sema_t * sema, const char * name) {
80 scope_t * scope = sema->current_scope;
81 while (scope) {
82 symbol_t * sym = scope->symbols;
83 while (sym) {
84 if (strcmp(sym->name, name) == 0)
85 return sym;
86 sym = sym->next;
87 }
88 scope = scope->parent;
89 }
90 return NULL;

205static void compile_expression(codegen_t * cg, ast_node_t * node) {
206 if (!node)
207 return;
208
209 switch (node->type) {
210 case AST_INTEGER_LITERAL:
211 emit_instruction(cg, OP_LOAD_INT, (int)node->literal.int_val, node->line);
212 break;
213
214 case AST_FLOAT_LITERAL:
215 emit_instruction(cg, OP_LOAD_FLOAT, (int)node->literal.float_val, node->line);
216 break;
217
218 case AST_STRING_LITERAL:
219 {
220 constant_t * c = (constant_t *)calloc(1, sizeof(constant_t));
221 c->type.kind = TYPE_STRING;
222 c->string_val = node->literal.string_val;
223 if (cg->current_function->constants_tail)
224 cg->current_function->constants_tail->next = c;
225 else
226 cg->current_function->constants = c;
227 cg->current_function->constants_tail = c;
228 emit_instruction(cg, OP_LOAD_STRING, 0, node->line);
229 break;
230 }
231
232 case AST_BOOL_LITERAL:
233 if (node->literal.int_val)
234 emit_instruction(cg, OP_LOAD_TRUE, 0, node->line);
235 else
236 emit_instruction(cg, OP_LOAD_FALSE, 0, node->line);
237 break;
238
239 case AST_NULL_LITERAL:
240 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
241 break;
242
243 case AST_IDENTIFIER:
244 {
245 symbol_t * sym = sema_lookup(cg->sema, node->identifier.name);
246 if (sym) {
247 if (sym->depth == 0)
248 emit_instruction(cg, OP_LOAD_GLOBAL, sym->stack_offset, node->line);
249 else
250 emit_instruction(cg, OP_LOAD_LOCAL, sym->stack_offset, node->line);
251 }
252 break;
253 }
254
255 case AST_BINARY_EXPR:
256 compile_expression(cg, node->binary_expr.left);
257 compile_expression(cg, node->binary_expr.right);
258 compile_binary_op(cg, node->binary_expr.operator, node->line);
259 break;
260
261 case AST_UNARY_EXPR:
262 compile_expression(cg, node->unary_expr.operand);
263 if (node->unary_expr.operator== TOKEN_MINUS)
264 emit_instruction(cg, OP_NEG, 0, node->line);
265 else if (node->unary_expr.operator== TOKEN_BANG)
266 emit_instruction(cg, OP_NOT, 0, node->line);
267 break;
268
269 case AST_CALL_EXPR:
270 // Push arguments in reverse order
271 if (node->call_expr.arguments) {
272 ast_node_t * args = node->call_expr.arguments;
273 int argc = 0;
274 ast_node_t * arg;
275 for (arg = args; arg; arg = arg->next)
276 argc++;
277
278 for (int i = argc - 1; i >= 0; i--) {
279 arg = args;
280 for (int j = 0; j < i; j++)
281 arg = arg->next;
282 compile_expression(cg, arg);
283 }
284 }
285 compile_expression(cg, node->call_expr.callee);
286 emit_instruction(cg, OP_CALL, 0, node->line);
287 break;
288
289 case AST_ASSIGNMENT:
290 compile_expression(cg, node->assignment.value);
291 emit_instruction(cg, OP_DUP, 0, node->line);
292 if (node->assignment.target->type == AST_IDENTIFIER) {
293 symbol_t * sym = sema_lookup(cg->sema, node->assignment.target->identifier.name);
294 if (sym) {
295 if (sym->depth == 0)
296 emit_instruction(cg, OP_STORE_GLOBAL, sym->stack_offset, node->line);
297 else
298 emit_instruction(cg, OP_STORE_LOCAL, sym->stack_offset, node->line);
299 }
300 }
301 break;
302
303 case AST_ARRAY_LITERAL:
304 {
305 if (node->array_literal.elements) {
306 ast_node_t * elem = node->array_literal.elements;
307 int count = 0;
308 while (elem) {
309 count++;
310 elem = elem->next;
311 }
312
313 emit_instruction(cg, OP_LOAD_INT, count, node->line);
314 emit_instruction(cg, OP_NEW_ARRAY, 0, node->line);
315
316 elem = node->array_literal.elements;
317 int idx = 0;
318 while (elem) {
319 emit_instruction(cg, OP_DUP, 0, node->line);
320 emit_instruction(cg, OP_LOAD_INT, idx, node->line);
321 compile_expression(cg, elem);
322 emit_instruction(cg, OP_STORE_INDEX, 0, node->line);
323 emit_instruction(cg, OP_POP, 0, node->line);
324 elem = elem->next;
325 idx++;
326 }
327 }
328 else {
329 emit_instruction(cg, OP_LOAD_INT, 0, node->line);
330 emit_instruction(cg, OP_NEW_ARRAY, 0, node->line);
331 }
332 break;
333 }
334
335 default:
336 break;
337 }
338}

354static void compile_statement(codegen_t * cg, ast_node_t * node) {
355 if (!node)
356 return;
357
358 switch (node->type) {
359 case AST_VAR_DECL:
360 {
361 if (node->var_decl.initializer)
362 compile_expression(cg, node->var_decl.initializer);
363 else
364 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
365
366 symbol_t * sym = sema_define(cg->sema, node->var_decl.name, SYM_VARIABLE, NULL);
367 emit_instruction(cg, OP_STORE_LOCAL, sym->stack_offset, node->line);
368 emit_instruction(cg, OP_POP, 0, node->line);
369 break;
370 }
371
372 case AST_EXPR_STMT:
373 compile_expression(cg, node);
374 emit_instruction(cg, OP_POP, 0, node->line);
375 break;
376
377 case AST_IF_STMT:
378 {
379 c23_maybe_unused int * jumps = (int *)malloc(sizeof(int) * 16);
380 c23_maybe_unused int jump_count = 0;
381
382 compile_expression(cg, node->if_stmt.condition);
383 int else_jump = cg->instruction_count;
384 emit_instruction(cg, OP_JUMP_IF_FALSE, 0, node->line);
385
386 compile_block(cg, node->if_stmt.then_branch);
387 int end_jump = cg->instruction_count;
388 emit_instruction(cg, OP_JUMP, 0, node->line);
389
390 // Patch else jump
391 instruction_t * inst = cg->current_function->code;
392 int idx = 0;
393 while (inst && idx < else_jump)
394 inst = inst->next;
395 if (inst)
396 inst->arg = cg->instruction_count;
397
398 // Elif chains
399 ast_node_t * elif = node->if_stmt.elif_list;
400 while (elif) {
401 compile_expression(cg, elif->if_stmt.condition);
402 c23_maybe_unused int elif_jump = cg->instruction_count;
403 emit_instruction(cg, OP_JUMP_IF_FALSE, 0, node->line);
404 compile_block(cg, elif->if_stmt.then_branch);
405 emit_instruction(cg, OP_JUMP, end_jump, node->line);
406 elif = elif->next;
407 }
408
409 // Else branch
410 if (node->if_stmt.else_branch)
411 compile_block(cg, node->if_stmt.else_branch);
412
413 // Patch end jump
414 inst = cg->current_function->code;
415 idx = 0;
416 while (inst && idx < end_jump)
417 inst = inst->next;
418 if (inst)
419 inst->arg = cg->instruction_count;
420
421 free(jumps);
422 break;
423 }
424
425 case AST_WHILE_STMT:
426 {
427 int loop_start = cg->instruction_count;
428
429 compile_expression(cg, node->while_stmt.condition);
430 int exit_jump = cg->instruction_count;
431 emit_instruction(cg, OP_JUMP_IF_FALSE, 0, node->line);
432
433 compile_block(cg, node->while_stmt.body);
434 emit_instruction(cg, OP_JUMP, loop_start, node->line);
435
436 // Patch exit jump
437 instruction_t * inst = cg->current_function->code;
438 int idx = 0;
439 while (inst && idx < exit_jump)
440 inst = inst->next;
441 if (inst)
442 inst->arg = cg->instruction_count;
443 break;
444 }
445
446 case AST_FOR_STMT:
447 {
448 if (node->for_stmt.initializer)
449 compile_statement(cg, node->for_stmt.initializer);
450
451 int loop_start = cg->instruction_count;
452 int exit_jump = -1;
453
454 if (node->for_stmt.condition) {
455 compile_expression(cg, node->for_stmt.condition);
456 exit_jump = cg->instruction_count;
457 emit_instruction(cg, OP_JUMP_IF_FALSE, 0, node->line);
458 }
459
460 compile_block(cg, node->for_stmt.body);
461
462 if (node->for_stmt.increment) {
463 compile_expression(cg, node->for_stmt.increment);
464 emit_instruction(cg, OP_POP, 0, node->line);
465 }
466
467 emit_instruction(cg, OP_JUMP, loop_start, node->line);
468
469 if (exit_jump >= 0) {
470 instruction_t * inst = cg->current_function->code;
471 int idx = 0;
472 while (inst && idx < exit_jump)
473 inst = inst->next;
474 if (inst)
475 inst->arg = cg->instruction_count;
476 }
477 break;
478 }
479
480 case AST_DO_WHILE_STMT:
481 {
482 int loop_start = cg->instruction_count;
483
484 compile_block(cg, node->do_while_stmt.body);
485 compile_expression(cg, node->do_while_stmt.condition);
486 emit_instruction(cg, OP_JUMP_IF_TRUE, loop_start, node->line);
487 break;
488 }
489
490 case AST_RETURN_STMT:
491 if (node->return_stmt.value)
492 compile_expression(cg, node->return_stmt.value);
493 else
494 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
495 emit_instruction(cg, OP_RETURN, 0, node->line);
496 break;
497
498 case AST_BREAK_STMT:
499 emit_instruction(cg, OP_JUMP, -1, node->line); // TODO: resolve break target
500 break;
501
502 case AST_CONTINUE_STMT:
503 emit_instruction(cg, OP_JUMP, -1, node->line); // TODO: resolve continue target
504 break;
505
506 case AST_TRY_STMT:
507 {
508 c23_maybe_unused int try_start = cg->instruction_count;
509 compile_block(cg, node->try_stmt.try_block);
510 int try_end = cg->instruction_count;
511
512 // Catch block
513 if (node->try_stmt.catch_block) {
514 sema_push_scope(cg->sema, "catch");
515 if (node->try_stmt.catch_var)
516 sema_define(cg->sema, node->try_stmt.catch_var, SYM_VARIABLE, NULL);
517 compile_block(cg, node->try_stmt.catch_block);
518 sema_pop_scope(cg->sema);
519 }
520
521 // Finally block
522 if (node->try_stmt.finally_block)
523 compile_block(cg, node->try_stmt.finally_block);
524
525 emit_instruction(cg, OP_CATCH, try_end, node->line);
526 break;
527 }
528
529 case AST_THROW_STMT:
530 compile_expression(cg, node->throw_stmt.exception);
531 emit_instruction(cg, OP_THROW, 0, node->line);
532 break;
533
534 case AST_BLOCK:
535 compile_block(cg, node);
536 break;
537
538 case AST_YIELD_STMT:
539 if (node->yield_stmt.value)
540 compile_expression(cg, node->yield_stmt.value);
541 else
542 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
543 emit_instruction(cg, OP_YIELD, 0, node->line);
544 break;
545
546 case AST_FUNCTION_DECL:
547 {
548 // Save current function
549 function_t * parent_fn = cg->current_function;
550 instruction_t * parent_tail = cg->current_instruction;
551
552 // Create new function
553 cg->current_function = (function_t *)calloc(1, sizeof(function_t));
554 cg->current_function->name = strdup(node->function_decl.name);
555 cg->current_instruction = NULL;
556
557 // Define symbol
558 sema_define(cg->sema, node->function_decl.name, SYM_FUNCTION, NULL);
559
560 // Compile parameters
561 sema_push_scope(cg->sema, node->function_decl.name);
562 ast_node_t * param = node->function_decl.params;
563 while (param) {
564 sema_define(cg->sema, param->param.name, SYM_PARAMETER, NULL);
565 param = param->next;
566 }
567
568 // Compile body
569 if (node->function_decl.body)
570 compile_statement(cg, node->function_decl.body);
571
572 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
573 emit_instruction(cg, OP_RETURN, 0, node->line);
574
575 sema_pop_scope(cg->sema);
576
577 // Link function
578 cg->current_function->next = parent_fn;
579 cg->current_function = parent_fn;
580 cg->current_instruction = parent_tail;
581 break;
582 }
583
584 case AST_FIBER_EXPR:
585 // Compile fiber body as a function
586 if (node->fiber_expr.body)
587 compile_block(cg, node->fiber_expr.body);
588 emit_instruction(cg, OP_LOAD_NULL, 0, node->line);
589 emit_instruction(cg, OP_RETURN, 0, node->line);
590 emit_instruction(cg, OP_CREATE_FIBER, 0, node->line);
591 break;
592
593 default:
594 compile_expression(cg, node);
595 emit_instruction(cg, OP_POP, 0, node->line);
596 break;
597 }
598}

656vm_object_t * vm_execute(vm_t * vm, function_t * entry) {
657 if (!entry || !entry->code)
658 return NULL;
659
660 uint8_t * ip = (uint8_t *)entry->code;
661 vm_frame_t frame;
662 frame.function = entry;
663 frame.ip = ip;
664 frame.locals = (uint8_t *)calloc(1, 1024);
665 frame.stack_base = vm->stack_top;
666
667 vm->frames = &frame;
668 vm->frame_count = 1;
669
670 while (true) {
671 opcode_t op = (opcode_t)*ip++;
672
673 switch (op) {
674 case OP_HALT:
675 goto done;
676
677 case OP_LOAD_INT:
678 push_int(vm, (int64_t)*(int32_t *)ip);
679 ip += 4;
680 break;
681
682 case OP_LOAD_NULL:
683 push_int(vm, 0);
684 break;
685
686 case OP_LOAD_TRUE:
687 push_int(vm, 1);
688 break;
689
690 case OP_LOAD_FALSE:
691 push_int(vm, 0);
692 break;
693
694 case OP_ADD:
695 {
696 int64_t b = pop_int(vm);
697 int64_t a = pop_int(vm);
698 push_int(vm, a + b);
699 break;
700 }
701
702 case OP_SUB:
703 {
704 int64_t b = pop_int(vm);
705 int64_t a = pop_int(vm);
706 push_int(vm, a - b);
707 break;
708 }
709
710 case OP_MUL:
711 {
712 int64_t b = pop_int(vm);
713 int64_t a = pop_int(vm);
714 push_int(vm, a * b);
715 break;
716 }
717
718 case OP_DIV:
719 {
720 int64_t b = pop_int(vm);
721 int64_t a = pop_int(vm);
722 push_int(vm, a / b);
723 break;
724 }
725
726 case OP_MOD:
727 {
728 int64_t b = pop_int(vm);
729 int64_t a = pop_int(vm);
730 push_int(vm, a % b);
731 break;
732 }
733
734 case OP_NEG:
735 {
736 int64_t a = pop_int(vm);
737 push_int(vm, -a);
738 break;
739 }
740
741 case OP_NOT:
742 {
743 int64_t a = pop_int(vm);
744 push_int(vm, !a);
745 break;
746 }
747
748 case OP_CMP_EQ:
749 {
750 int64_t b = pop_int(vm);
751 int64_t a = pop_int(vm);
752 push_int(vm, a == b);
753 break;
754 }
755
756 case OP_CMP_NE:
757 {
758 int64_t b = pop_int(vm);
759 int64_t a = pop_int(vm);
760 push_int(vm, a != b);
761 break;
762 }
763
764 case OP_CMP_LT:
765 {
766 int64_t b = pop_int(vm);
767 int64_t a = pop_int(vm);
768 push_int(vm, a < b);
769 break;
770 }
771
772 case OP_CMP_LE:
773 {
774 int64_t b = pop_int(vm);
775 int64_t a = pop_int(vm);
776 push_int(vm, a <= b);
777 break;
778 }
779
780 case OP_CMP_GT:
781 {
782 int64_t b = pop_int(vm);
783 int64_t a = pop_int(vm);
784 push_int(vm, a > b);
785 break;
786 }
787
788 case OP_CMP_GE:
789 {
790 int64_t b = pop_int(vm);
791 int64_t a = pop_int(vm);
792 push_int(vm, a >= b);
793 break;
794 }
795
796 case OP_AND:
797 {
798 int64_t b = pop_int(vm);
799 int64_t a = pop_int(vm);
800 push_int(vm, a & b);
801 break;
802 }
803
804 case OP_OR:
805 {
806 int64_t b = pop_int(vm);
807 int64_t a = pop_int(vm);
808 push_int(vm, a | b);
809 break;
810 }
811
812 case OP_JUMP:
813 {
814 int32_t offset = *(int32_t *)ip;
815 ip = (uint8_t *)(uintptr_t)offset; // TODO: properly resolve jump targets
816 break;
817 }
818
819 case OP_JUMP_IF_TRUE:
820 {
821 int64_t val = pop_int(vm);
822 if (val) {
823 int32_t offset = *(int32_t *)ip;
824 ip = (uint8_t *)(uintptr_t)offset;
825 }
826 else {
827 ip += 4;
828 }
829 break;
830 }
831
832 case OP_JUMP_IF_FALSE:
833 {
834 int64_t val = pop_int(vm);
835 if (!val) {
836 int32_t offset = *(int32_t *)ip;
837 ip = (uint8_t *)(uintptr_t)offset;
838 }
839 else {
840 ip += 4;
841 }
842 break;
843 }
844
845 case OP_RETURN:
846 {
847 goto done;
848 }
849
850 case OP_POP:
851 vm->stack_top -= sizeof(int64_t);
852 break;
853
854 case OP_DUP:
855 push_int(vm, *(int64_t *)(vm->stack_top - sizeof(int64_t)));
856 break;
857
858 case OP_NEW_ARRAY:
859 {
860 int64_t count = pop_int(vm);
861 vm_object_t * arr = alloc_object(vm, TYPE_ARRAY, sizeof(vm_object_t));
862 arr->array.data = (uint64_t *)calloc(count, sizeof(uint64_t));
863 arr->array.count = count;
864 arr->array.capacity = count;
865 *(vm_object_t **)vm->stack_top = arr;
866 vm->stack_top += sizeof(vm_object_t *);
867 break;
868 }
869
870 case OP_STORE_INDEX:
871 {
872 int64_t idx = pop_int(vm);
873 vm_object_t * arr = *(vm_object_t **)(vm->stack_top - sizeof(vm_object_t *) - sizeof(int64_t));
874 int64_t val = pop_int(vm);
875 if (arr && arr->type.kind == TYPE_ARRAY && idx >= 0 && (size_t)idx < arr->array.count)
876 arr->array.data[idx] = val;
877 break;
878 }
879
880 case OP_DEBUG:
881 {
882 int64_t val = pop_int(vm);
883 printf("DEBUG: %lld\n", (long long)val);
884 break;
885 }
886
887 default:
888 break;
889 }
890 }
891
892done:
893 free(frame.locals);
894
895 if (vm->stack_top > vm->stack) {
896 vm_object_t * result = *(vm_object_t **)(vm->stack_top - sizeof(vm_object_t *));
897 return result;
898 }
899
900 return NULL;
901}

914bool infix_compile(infix_compiler_t * compiler, const char * source) {
915 lexer_t * lexer = lexer_create(source);
916 parser_t * parser = parser_create(lexer);
917
918 ast_node_t * ast = parser_parse(parser);
919
920 if (parser_had_error(parser)) {
921 fprintf(stderr, "Parse error: %s\n", parser_get_error(parser));
922 parser_destroy(parser);
923 return false;
924 }
925
926 sema_t * sema = sema_create();
927 codegen_t * cg = codegen_create(sema);
928
929 c23_maybe_unused function_t * entry = codegen_compile(cg, ast);
930
931 if (codegen_had_error(cg)) {
932 fprintf(stderr, "Codegen error: %s\n", codegen_get_error(cg));
933 codegen_destroy(cg);
934 sema_destroy(sema);
935 parser_destroy(parser);
936 ast_free_tree(ast);
937 return false;
938 }
939
940 compiler->lexer = lexer;
941 compiler->parser = parser;
942 compiler->sema = sema;
943 compiler->codegen = cg;
944
945 return true;
946}

954
