208static uint8_t * build_code_with_exitcall(
209 uint64_t return_value, int arch, uint32_t exitprocess_iat_rva, uint32_t text_rva, size_t * out_size) {
210 size_t size = 64; // Safe buffer
211 uint8_t * buf = (uint8_t *)calloc(1, size);
212 if (!buf)
213 return NULL;
214 size_t pos = 0;
215 if (arch == EMIT_ARCH_X86_64) {
216 buf[pos++] = 0x48;
217 buf[pos++] = 0x83;
218 buf[pos++] = 0xEC;
219 buf[pos++] = 0x28; // sub rsp, 40
220 buf[pos++] = 0x48;
221 buf[pos++] = 0xB9; // mov rcx, imm64
222 *(uint64_t *)(buf + pos) = return_value;
223 pos += 8;
224 buf[pos++] = 0xFF;
225 buf[pos++] = 0x15; // call [rip+disp]
226 uint32_t rip = text_rva + (uint32_t)pos + 4;
227 *(uint32_t *)(buf + pos) = (uint32_t)(exitprocess_iat_rva - rip);
228 pos += 4;
229 buf[pos++] = 0xC3;
230 }
231 else if (arch == EMIT_ARCH_AARCH64) {
232 /* mov x0, #return_value (limited to 16-bit for simplicity in this stub) */
233 uint32_t mov_x0 = 0xD2800000 | ((return_value & 0xFFFF) << 5) | 0;
234 *(uint32_t *)(buf + pos) = mov_x0;
235 pos += 4;
236 /* adrp x8, exitprocess_iat_rva */
237 uint32_t adrp = 0x90000008;
238 int64_t diff = (int64_t)exitprocess_iat_rva - (int64_t)text_rva;
239 int64_t pagediff = (diff >> 12);
240 adrp |= ((pagediff & 0x3) << 29) | ((pagediff & 0x1FFFFC) << 3);
241 *(uint32_t *)(buf + pos) = adrp;
242 pos += 4;
243 /* ldr x8, [x8, #:lo12:exitprocess_iat_rva] */
244 uint32_t ldr = 0xF9400108 | ((exitprocess_iat_rva & 0xFFF) >> 3) << 10;
245 *(uint32_t *)(buf + pos) = ldr;
246 pos += 4;
247 /* blr x8 */
248 *(uint32_t *)(buf + pos) = 0xD63F0100;
249 pos += 4;
250 }
251 else {
252 buf[pos++] = 0x68; // push imm32
253 *(uint32_t *)(buf + pos) = (uint32_t)return_value;
254 pos += 4;
255 buf[pos++] = 0xFF;
256 buf[pos++] = 0x15; // call [addr]
257 *(uint32_t *)(buf + pos) = 0x00400000 + exitprocess_iat_rva;
258 pos += 4;
259 }
260 *out_size = pos;
261 return buf;
262}
