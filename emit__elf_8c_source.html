84 size_t total = 1;
85 for (size_t i = 1; i < num_strings; i++)

103static uint8_t * build_symtab(
104 emit_symbol_t * symbols, size_t num_symbols, const char * strtab_buf, size_t strtab_sz, size_t * out_size) {
105 size_t total = num_symbols * sizeof(Elf64_Sym);
106 uint8_t * buf = calloc(1, total);
107 if (!buf)
108 return NULL;
109 Elf64_Sym * sym = (Elf64_Sym *)buf;
110 size_t sym_idx = 1;
111 emit_symbol_t * s = symbols;
112 while (s && sym_idx < num_symbols) {
113 if (s->name) {
114 const char * match = strstr(strtab_buf, s->name);
115 if (match)
116 sym[sym_idx].st_name = (uint32_t)(match - strtab_buf);
117 }
118 sym[sym_idx].st_info = ELF64_ST_INFO(STB_GLOBAL, s->is_function ? STT_FUNC : STT_OBJECT);
119 sym[sym_idx].st_value = s->value;
120 s = s->next;
121 sym_idx++;
122 }
123 *out_size = total;
124 return buf;
125}

127static uint8_t * build_rela(emit_relocation_t * relocs, emit_symbol_t * symbols, size_t * out_count) {
128 size_t num = 0;
129 emit_relocation_t * r = relocs;
130 while (r) {
131 num++;
132 r = r->next;
133 }
134 if (num == 0) {
135 *out_count = 0;
136 return NULL;
137 }
138 uint8_t * buf = calloc(num, sizeof(Elf64_Rela));
139 Elf64_Rela * ent = (Elf64_Rela *)buf;
140 r = relocs;
141 for (size_t i = 0; i < num && r; i++, r = r->next) {
142 ent[i].r_offset = r->offset;
143 size_t idx = 1;
144 emit_symbol_t * s = symbols;
145 while (s) {
146 if (strcmp(s->name, r->symbol_name) == 0)
147 break;
148 idx++;
149 s = s->next;
150 }
151 ent[i].r_sym = (uint32_t)idx;
152 ent[i].r_type = ELF_R_X86_64_PC32;
153 }
154 *out_count = num;
155 return buf;
156}

289 free(shdrs);
290 free(shstr_buf);

292 free(symtab);

294 *out_data = buf;
295 *out_size = total;

300 if (!ctx || !out_data || !out_size)

302 size_t code_sz = 0;
303 for (emit_section_t * s = ctx->sections; s; s = s->next)
304 code_sz = ELF_ALIGN(code_sz, 16) + s->size;
305 size_t total = ELF_ALIGN(0x1000 + code_sz, 4096);
306 uint8_t * buf = calloc(1, total);
307 Elf64_Ehdr * ehdr = (Elf64_Ehdr *)buf;

309 ehdr->e_ident[4] = ELFCLASS64;

312 ehdr->e_type = ET_EXEC;
313 ehdr->e_machine = elf_get_machine(ctx->arch);
314 ehdr->e_version = EV_CURRENT;
315 ehdr->e_entry = 0x401000 + (ctx->symbols ? ctx->symbols->value : 0);
316 ehdr->e_phoff = sizeof(Elf64_Ehdr);
317 ehdr->e_ehsize = sizeof(Elf64_Ehdr);
318 ehdr->e_phentsize = sizeof(Elf64_Phdr);
319 ehdr->e_phnum = 1;
320 Elf64_Phdr * phdr = (Elf64_Phdr *)(buf + sizeof(Elf64_Ehdr));
321 phdr->p_type = PT_LOAD;
322 phdr->p_flags = PF_R | PF_X;
323 phdr->p_vaddr = 0x400000;
324 phdr->p_paddr = 0x400000;
325 phdr->p_filesz = total;
326 phdr->p_memsz = total;
327 phdr->p_align = 0x1000;
328 size_t off = 0x1000;
329 for (emit_section_t * s = ctx->sections; s; s = s->next) {
330 off = ELF_ALIGN(off, 16);
331 memcpy(buf + off, s->data, s->size);
332 off += s->size;
